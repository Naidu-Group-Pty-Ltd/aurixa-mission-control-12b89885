name: Decommission builder portal functions

# Phase 7 of the network extraction (docs/builder-portal/45-network-extraction-plan.md
# §7): the Builder / Developer Portal moved to the Builders Network, /builder/*
# redirects there, and this workflow deletes the portal's LIVE deployments from
# the Supabase project — dormant surface is exactly what the plan's "drop,
# don't leave dormant" is about. Wave 1 deleted the sixteen portal-session
# functions; wave 3 dropped the builder schema, which turned the read-only
# admin family and the image-pipeline workers into deployments that can only
# error, so the list WIDENED to cover them (the reviewed edit the first
# version of this file said such a widening must be). It deletes on demand,
# and nothing else:
#
# - It is workflow_dispatch only. Deleting a deployed function is a deliberate
#   act with a name in the Actions log, never a side effect of a push.
# - It can delete ONLY the names in RETIRED below. A deletion workflow that
#   can delete arbitrary functions is a hazard this file must not be;
#   widening it further is another reviewed edit to the list.
# - A function already absent is a no-op, not a failure: re-running after a
#   partial run must finish the job.
# - The project ref resolves exactly as the deploy workflow's does (variable,
#   then this repo's own config.toml preamble, never a literal) so a clone
#   that inherits this file deletes from ITS OWN project and nobody else's.
#
# builder-stock-marketplace (the Command Centre read, on the network mirror
# since wave 2) and builder-network-inbound (the sync door) are NOT in the
# list and stay deployed. builder-stock-inpaint-probe is listed although no
# repository directory ever carried it — the deployment exists, and the
# deployment is what this deletes.

on:
  workflow_dispatch:
    inputs:
      functions:
        description: 'Space-separated subset of the retired list. Empty = the whole list.'
        required: false
        default: ''

concurrency:
  group: decommission-builder-portal-functions
  cancel-in-progress: false

jobs:
  decommission:
    runs-on: ubuntu-latest
    # The production environment. Where the owner restricts it to main and keeps
    # the deploy token in it alone, GitHub refuses any other ref before a step
    # runs, and a copy of this file on a branch cannot read the token
    # (docs/operations/DEPLOYMENT_PROTECTION.md). Reviewers are the owner's
    # setting; with none, this runs on merge as it always has.
    environment: supabase-production
    steps:
      # The accident guard. A dispatch runs the copy of this file on the ref it
      # names, so this stops a mistaken choice in "Run workflow" and nothing
      # more: the environment above is what a modified copy cannot remove.
      - name: Refuse any branch but main
        env:
          REF: ${{ github.ref }}
        run: |
          if [ "$REF" != 'refs/heads/main' ]; then
            echo "::error title=Decommission refused — not main::Functions are decommissioned only from main. This run was started on $REF, so nothing was deleted. Merge the change, then dispatch this workflow on main."
            exit 1
          fi

      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

      - name: refuse to pretend
        env:
          TOKEN: ${{ secrets.SUPABASE_ACCESS_TOKEN }}
        run: |
          if [ -z "${TOKEN:-}" ]; then
            echo '::error::No SUPABASE_ACCESS_TOKEN secret is configured — nothing was deleted. Add it to the supabase-production environment (Settings → Environments), then re-run.'
            exit 1
          fi

      - name: Resolve this deployment's project
        id: project
        env:
          PROJECT_REF_VAR: ${{ vars.SUPABASE_PROJECT_REF }}
        run: |
          set -uo pipefail
          ref="${PROJECT_REF_VAR:-}"
          if [ -z "$ref" ] && [ -f supabase/config.toml ]; then
            ref=$(sed -n '/^\[/q; s/^[[:space:]]*project_id[[:space:]]*=[[:space:]]*"\([a-z]\{20\}\)".*/\1/p' \
              supabase/config.toml | head -1)
          fi
          if [ -z "$ref" ]; then
            echo '::error::no project ref could be resolved for this repository — nothing was deleted.'
            exit 1
          fi
          echo "deleting from project $ref"
          echo "ref=$ref" >> "$GITHUB_OUTPUT"

      # Installed from Supabase's own release, not through `supabase/setup-cli`:
      # the Actions policy refuses a workflow naming a publisher other than
      # GitHub before a job starts (CI_SCOPE.md, "Every action is pinned to a
      # commit"). The tarball is checked against its sha256 before it is unpacked.
      - name: Install the Supabase CLI
        env:
          SUPABASE_CLI_VERSION: 2.121.0
          SUPABASE_CLI_SHA256: 4b481189efbdc57cf2d37bdee5f1a0d29574314398564b32a6dfb76c0fcf275f
        run: |
          set -euo pipefail
          curl -fsSL --retry 3 -o "$RUNNER_TEMP/supabase.tar.gz" \
            "https://github.com/supabase/cli/releases/download/v${SUPABASE_CLI_VERSION}/supabase_linux_amd64.tar.gz"
          echo "${SUPABASE_CLI_SHA256}  $RUNNER_TEMP/supabase.tar.gz" | sha256sum -c -
          mkdir -p "$HOME/.supabase/bin"
          tar -xzf "$RUNNER_TEMP/supabase.tar.gz" -C "$HOME/.supabase/bin" supabase
          echo "$HOME/.supabase/bin" >> "$GITHUB_PATH"
          "$HOME/.supabase/bin/supabase" --version

      - name: delete the retired portal functions
        env:
          SUPABASE_ACCESS_TOKEN: ${{ secrets.SUPABASE_ACCESS_TOKEN }}
          REQUESTED: ${{ inputs.functions }}
          PROJECT_REF: ${{ steps.project.outputs.ref }}
        run: |
          set -uo pipefail
          RETIRED='builder-portal-login builder-portal-logout builder-portal-verify builder-portal-accept-invite builder-portal-forgot-password builder-portal-reset-password builder-portal-change-password builder-portal-invite builder-portal-projects builder-portal-inventory builder-portal-transactions builder-portal-construction builder-portal-delivery builder-portal-collaboration builder-portal-workspace builder-portal-stock builder-portal-admin builder-projects-admin builder-inventory-admin builder-transactions-admin builder-construction-admin builder-delivery-admin builder-collaboration-admin builder-workspace-admin builder-stock-image-settler builder-stock-inpaint-probe builder-document-processor builder-stock-link-callback'
          list="${REQUESTED:-$RETIRED}"
          failed=""
          for fn in $list; do
            case " $RETIRED " in
              *" $fn "*) ;;
              *)
                echo "::error::'$fn' is not on the retired list — this workflow deletes only the retired builder portal functions."
                exit 1
                ;;
            esac
          done
          for fn in $list; do
            echo "::group::delete $fn"
            out=$(supabase functions delete "$fn" --project-ref "$PROJECT_REF" 2>&1) && status=0 || status=$?
            echo "$out"
            if [ "$status" -ne 0 ]; then
              # Already gone is the outcome this workflow exists to produce.
              if printf '%s' "$out" | grep -qiE 'not[ _-]?found|does not exist|404'; then
                echo "$fn was already absent"
              else
                failed="$failed $fn"
              fi
            fi
            echo "::endgroup::"
          done
          if [ -n "$failed" ]; then
            echo "::error::deletion failed for:$failed"
            exit 1
          fi
